// Counts skeletons for Brokk (DwarfQuest=1), up to 4
01	Get.Value	ATTRIBUTE	DwarfQuest	Variable.1$
02	>Conditional	$Variable.1=1	08
03	Get.Value	ATTRIBUTE	SkeletonsSlain	Variable.2$
04	>Conditional	$Variable.2<4	08
05	Update.Attribute	SkeletonsSlain	+1
06	Get.Value	CALCULATION	$Variable.2+1	Variable.3$
07	Display.Message	Yellow	Skeletons smashed for Brokk: $Variable.3 of 4.
08	End
